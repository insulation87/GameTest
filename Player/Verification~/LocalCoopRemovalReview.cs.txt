public static class LocalCoopRemovalReview
{

    const BindingFlags Flags = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
    static readonly List<string> results = new List<string>(), errors = new List<string>();
    static readonly Stack<IEnumerator> steps = new Stack<IEnumerator>();
    static double next, deadline;
    static string Out => Path.GetFullPath(Path.Combine(Application.dataPath,"../.."));
    static Player player;
    static PlayerHealth health;
    static PlayerShooter shooter;
    static WeaponInventory inventory;
    static RogueTactics tactics;

    static Vector3 arena = new Vector3(10000,0,10000);

    static LocalCoopRemovalReview() { EditorApplication.playModeStateChanged += Changed; }
    public static void Run()
    {
        if (!Application.isBatchMode || !Path.GetFullPath(Application.dataPath).StartsWith(Path.GetTempPath(),StringComparison.OrdinalIgnoreCase))
            throw new Exception("Temporary batch copy required.");
        PlayerSettings.companyName="CodexIsolatedReview";
        PlayerSettings.productName="LocalCoopRemovalReview-"+new DirectoryInfo(Out).Name;
        LoginSceneSetup.PlayFromCurrentScene();
        EditorSceneManager.OpenScene("Assets/Scenes/Map1.unity",OpenSceneMode.Single);
        foreach (var player in Object.FindObjectsByType<Player>(FindObjectsSortMode.None))
        {
            var save = player.GetComponent<GameProgressSave>() ?? player.gameObject.AddComponent<GameProgressSave>();
            save.enabled = false;
        }
        foreach (var spawner in Object.FindObjectsByType<NPCSpawner>(FindObjectsSortMode.None)) spawner.enabled = false;
        SessionState.SetBool(Pending,true); EditorApplication.EnterPlaymode();
    }
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    static void IsolateSaves()
    {
        if (!Application.isBatchMode || !SessionState.GetBool(Pending,false)) return;
        string path=Path.Combine(Out,"isolated-progress.json");
        foreach(var suffix in new[]{"",".bak",".tmp"}) if(File.Exists(path+suffix)) File.Delete(path+suffix);
        GameProgressSave.StartFresh=true;
        foreach(var actor in Object.FindObjectsByType<Player>(FindObjectsSortMode.None))
        {
            var save=actor.GetComponent<GameProgressSave>();
            save.StoragePath=path; save.enabled=true;
        }
    }
    static void Changed(PlayModeStateChange state)
    {
        if (state != PlayModeStateChange.EnteredPlayMode || !SessionState.GetBool(Pending,false)) return;
        SessionState.SetBool(Pending,false); results.Clear(); errors.Clear(); steps.Clear(); steps.Push(Checks());
        next = EditorApplication.timeSinceStartup + .3; deadline = EditorApplication.timeSinceStartup + 360;
        Application.logMessageReceived += Log; EditorApplication.update += Tick;
    }
    static void Log(string text,string stack,LogType type)
    {
        if (stack.Contains("UnityEditor.Search") || text.Contains("burst JIT compiler")) return;
        if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert) errors.Add(text + "\n" + stack);
    }
    static void Tick()
    {
        if (EditorApplication.timeSinceStartup < next) return;
        try
        {
            if (EditorApplication.timeSinceStartup > deadline) throw new TimeoutException("Play Mode review timeout");
            while (steps.Count > 0)
            {
                var step = steps.Peek();
                if (!step.MoveNext()) { steps.Pop(); continue; }
                if (step.Current is IEnumerator nested) { steps.Push(nested); continue; }
                next = EditorApplication.timeSinceStartup + (step.Current is float seconds ? seconds : .05); return;
            }
            Check(errors.Count == 0,"No unexpected game/runtime errors: " + string.Join("\n",errors)); Finish(0);
        }
        catch (Exception ex) { results.Add("FAIL " + ex); Finish(1); }
    }
    static void Finish(int code)
    {
        File.WriteAllText(Path.Combine(Out,"local-coop-removal-result.txt"),(code == 0 ? "PASS" : "FAIL") + "\n" + string.Join("\n",results) + "\n" + string.Join("\n",errors));
        Application.logMessageReceived -= Log; EditorApplication.update -= Tick; EditorApplication.Exit(code);
    }
    static void Check(bool value,string message) { if (!value) throw new Exception(message); results.Add("PASS " + message); File.WriteAllLines(Path.Combine(Out,"local-coop-removal-live.txt"),results); }
    static object Get(object target,string field) => target.GetType().GetField(field,Flags).GetValue(target);
    static void Set(object target,string field,object value) => target.GetType().GetField(field,Flags).SetValue(target,value);
    static object Call(object target,string method,params object[] args) => target.GetType().GetMethod(method,Flags).Invoke(target,args);
    static void CloseUpgrade()
    {
        int guard = 0; while (health.IsChoosingUpgrade && guard++ < 24) health.SelectCoopUpgrade(0);
        Time.timeScale = 1;
    }
    static void Move(Vector3 ground)
    {
        if (!player.TryGetTeleportPosition(ground,Quaternion.identity,out var pose)) throw new Exception("Unsafe test teleport: " + ground);
        player.Teleport(pose,Quaternion.identity); Physics.SyncTransforms();
    }
    static IEnumerator KillEncounter(TacticalExpedition expedition,int id)
    {
        var markers = (TacticalMarker[])Get(expedition,"markers");
        Move(markers[id].transform.position);
        yield return .3f; CloseUpgrade();
        if (expedition.ActiveEncounter < 0) expedition.TryBeginEncounter(id,player);
        Check(expedition.ActiveEncounter == id,"Actual Map1 encounter starts: " + id);
        for (int iteration = 0; iteration < 30 && !expedition.Progress.completed[id]; iteration++)
        {
            foreach (var enemy in expedition.EncounterEnemies.ToArray())
            {
                if (enemy == null || enemy.IsDead) continue;
                var boss = enemy.GetComponent<BossController>();
                if (boss != null && boss.IsTransforming) continue;
                enemy.TakeDamage(100000,health,enemy.transform.position - enemy.transform.forward*3);
            }
            Set(expedition,"nextWave",Time.time - 1);
            yield return id == 2 ? 1f : .25f; CloseUpgrade();
        }
        Check(expedition.Progress.completed[id],"Real death paths complete encounter and all waves: " + id);
    }
    static Keyboard keyboard;
    static void Keys(params Key[] values) { InputSystem.QueueStateEvent(keyboard,new KeyboardState(values)); InputSystem.Update(); }
    static IEnumerator WaitFor(Func<bool> condition)
    {
        double until=EditorApplication.timeSinceStartup+45;
        while(!condition()) { if(EditorApplication.timeSinceStartup>until)throw new TimeoutException("Waiting for scene transition"); yield return .05f; }
    }
    static IEnumerator Checks()
    {
        yield return 1f;
        Check(Application.isPlaying,"Feature removal is verified in actual Unity Play Mode");
        player=Object.FindFirstObjectByType<Player>(); health=player.GetComponent<PlayerHealth>(); inventory=player.GetComponent<WeaponInventory>();
        Check(Application.companyName=="CodexIsolatedReview" && (string)Get(player.GetComponent<GameProgressSave>(),"sessionPath")==Path.Combine(Out,"isolated-progress.json"),"Game progress and persistent preferences are isolated from user data");
        TacticalExpedition.Instance.enabled=false; CloseUpgrade(); Set(health,"invincibleUntil",Time.time+1000);
        Check(CoopSession.Mode==CoopMode.SinglePlayer && Object.FindObjectsByType<Player>(FindObjectsSortMode.None).Length==1,"Actual Map1 solo run contains exactly one player");
        Check(!Enum.GetNames(typeof(CoopMode)).Contains("SplitScreen") && (int)CoopMode.Host==2 && (int)CoopMode.Client==3,"Split-screen mode is removed while network enum IDs remain stable");
        keyboard=InputSystem.AddDevice<Keyboard>();
        var pad=InputSystem.AddDevice<Gamepad>();
        GameSettings.Load("LocalCoopRemovalReview.",false); GameSettings.Apply(GameSettings.Defaults(),false);
        Keys(Key.W); Check(CoopInput.Read(0,0).move.y==1,"Real primary keyboard input still reaches the gameplay frame");
        var before=player.transform.position; yield return .35f;
        Check(Vector3.Distance(before,player.transform.position)>.05f,"Primary keyboard still moves the actual solo player"); Keys();
        Keys(Key.UpArrow,Key.PageUp,Key.Numpad0);
        Check(CoopInput.Read(1,0).move==Vector2.zero && CoopInput.Read(1,0).held==0,"Second keyboard input no longer creates a second-player frame"); Keys();
        InputSystem.QueueStateEvent(pad,new GamepadState().WithButton(GamepadButton.RightShoulder)); InputSystem.Update();
        Check(CoopInput.Read(1,0).held==0 && CoopInput.Read(2,0).held==0,"A connected gamepad cannot reactivate the removed local player");
        Check(!GameControls.Available(1,GameAction.Skill) && !GameControls.Available(2,GameAction.Skill),"Retired second-player binding profiles are unavailable");
        Check(Object.FindObjectsByType<Player>(FindObjectsSortMode.None).Length==1,"Secondary keyboard and gamepad never create an extra actor");
        InputSystem.RemoveDevice(pad);
        var settingsCanvas=new GameObject("Settings Preview",typeof(RectTransform),typeof(Canvas),typeof(CanvasScaler));
        settingsCanvas.GetComponent<Canvas>().renderMode=RenderMode.ScreenSpaceOverlay; GameUITheme.FitCanvas(settingsCanvas.GetComponent<CanvasScaler>());
        var settings=GameSettingsMenu.Attach(settingsCanvas,()=>settingsCanvas.SetActive(false)); settings.SelectTab(3); yield return .1f;
        var bindingButtons=(Button[])Get(settings,"profiles");
        Check(bindingButtons.Length==1 && bindingButtons[0].GetComponentInChildren<Text>().text=="鍵盤與滑鼠","Real settings expose only the keyboard/mouse profile");
        Check(!settingsCanvas.GetComponentsInChildren<Text>(true).Any(t=>t.text.Contains("玩家 2") || t.text.Contains("玩家2")),"Second-player controls and help have been removed from the settings UI");
        settings.SelectBindingProfile(1); Check((int)Get(settings,"bindingProfile")==0,"Legacy settings calls cannot select a second-player profile");
        CaptureCanvas(settingsCanvas.GetComponent<Canvas>(),"local-coop-removal-settings.png");
        settings.BeginCapture(GameAction.Skill); settings.CaptureBinding((int)KeyCode.PageUp); settings.ApplyAndClose(); yield return .1f;
        Check(GameSettings.Current.bindings.playerOne[(int)GameAction.Skill]==(int)KeyCode.PageUp,"Player can bind a formerly reserved second-player key");
        GameSettings.Load("LocalCoopRemovalReview.",false);
        Check(GameSettings.Current.bindings.playerOne[(int)GameAction.Skill]==(int)KeyCode.PageUp,"Primary custom bindings survive save and reload without retired-profile conflicts");
        Keys(Key.PageUp); Check((CoopInput.Read(0,0).held&(int)CoopButton.Skill)!=0,"New primary binding reaches real input");
        var skills=player.GetComponent<CharacterSkills>();
        Check((CoopInput.Read(0,0).pressed&(int)CoopButton.Skill)!=0,"Rebound key produces a primary skill input frame");
        Check(skills.TryUseSkill() && skills.CooldownRemaining>0,"Actual single-player character skill remains usable after settings changes"); Keys();
        Object.DestroyImmediate(settingsCanvas);
        Check(player.GetComponent<GameProgressSave>().SaveNow(),"Solo progress can still be saved to the isolated file");
        Check(GameSceneFlow.Load(GameSceneFlow.LoginScene),"Actual login scene opens before changing account mode");
        yield return WaitFor(()=>!GameSceneFlow.IsLoading && Object.FindObjectsByType<Player>(FindObjectsSortMode.None).Length==0);
        Check(CloudAccountService.EnsureInstance().EnterGuest(),"Guest can enter the retained single-player and LAN menu flow");
        yield return WaitFor(()=>!GameSceneFlow.IsLoading && Object.FindFirstObjectByType<MainMenuController>()!=null);
        Check(!CoopSession.StartSplit("Map1","breaker","medic") && CoopSession.Mode==CoopMode.SinglePlayer && !GameSceneFlow.IsLoading,"Legacy split-screen launch is rejected without starting a scene or changing mode");
        bool rejected=false;
        try { CoopSession.Ensure().SpawnActor(1,"medic",true,1); } catch(InvalidOperationException){rejected=true;}
        Check(rejected,"Session refuses to spawn a second local actor through the old API");
        typeof(CoopSession).GetProperty("LastError").SetValue(null,null);
        Check(GameSceneFlow.Load(GameSceneFlow.MenuScene),"Real transition back to MainMenu starts");
        yield return WaitFor(()=>!GameSceneFlow.IsLoading && Object.FindFirstObjectByType<MainMenuController>()!=null);
        yield return .25f;
        var main=Object.FindFirstObjectByType<MainMenuController>();
        Check(!Object.FindObjectsByType<Text>(FindObjectsSortMode.None).Any(t=>t.text.Contains("同機雙人")),"Actual main menu no longer advertises local co-op");
        var entry=Object.FindObjectsByType<Button>(FindObjectsInactive.Include,FindObjectsSortMode.None).First(b=>b.name=="Multiplayer");
        entry.onClick.Invoke(); yield return .1f;
        var menu=Object.FindFirstObjectByType<MultiplayerMenu>();
        Check(menu!=null && menu.gameObject.activeInHierarchy,"Actual multiplayer menu opens through its main-menu button");
        Check(!menu.GetComponentsInChildren<Text>(true).Any(t=>t.text.Contains("玩家 2") || t.text.Contains("同機雙人") || t.text.Contains("分割畫面")),"Actual multiplayer UI has no second character, split-screen entry or related help");
        Check(((Button[])Get(menu,"launchButtons")).Length==2,"Only host and join launch actions remain");

        Check(GameSettings.Current.bindings.playerOne[(int)GameAction.Skill]==(int)KeyCode.PageUp,"Custom primary controls remain available through menu transitions");
        InputSystem.RemoveDevice(keyboard);
    }
    static void CaptureCanvas(Canvas canvas,string name)
    {
        var cameraNode=new GameObject("UI Review Camera",typeof(Camera)); var cam=cameraNode.GetComponent<Camera>();
        cam.clearFlags=CameraClearFlags.SolidColor; cam.backgroundColor=GameUITheme.Ink; cam.cullingMask=1<<31; cam.nearClipPlane=.1f; cam.farClipPlane=20;
        foreach(var node in canvas.GetComponentsInChildren<Transform>(true))node.gameObject.layer=31;
        canvas.renderMode=RenderMode.ScreenSpaceCamera; canvas.worldCamera=cam; canvas.planeDistance=1;
        var render=new RenderTexture(1920,1080,24); cam.targetTexture=render; Canvas.ForceUpdateCanvases(); cam.Render();
        var previous=RenderTexture.active; RenderTexture.active=render;
        var bitmap=new Texture2D(1920,1080,TextureFormat.RGB24,false); bitmap.ReadPixels(new Rect(0,0,1920,1080),0,0); bitmap.Apply();
        File.WriteAllBytes(Path.Combine(Out,name),bitmap.EncodeToPNG()); RenderTexture.active=previous; cam.targetTexture=null;
        canvas.renderMode=RenderMode.ScreenSpaceOverlay; canvas.worldCamera=null;
        Object.DestroyImmediate(bitmap); render.Release(); Object.DestroyImmediate(render); Object.DestroyImmediate(cameraNode);
    }
}
