    public static void Run()
    {
        if (!Application.isBatchMode || !Path.GetFullPath(Application.dataPath).StartsWith(Path.GetTempPath(),StringComparison.OrdinalIgnoreCase))
            throw new Exception("Temporary batch copy required.");
        PlayerSettings.companyName="CodexIsolatedReview";
        PlayerSettings.productName="FirstFloorKeyRemovalReview-"+new DirectoryInfo(Out).Name;
        LoginSceneSetup.PlayFromCurrentScene();
        EditorSceneManager.OpenScene("Assets/Scenes/Map1.unity",OpenSceneMode.Single);
        foreach (var player in Object.FindObjectsByType<Player>(FindObjectsSortMode.None))
        {
            var save = player.GetComponent<GameProgressSave>() ?? player.gameObject.AddComponent<GameProgressSave>();
            save.enabled = false;
        }
        foreach (var spawner in Object.FindObjectsByType<NPCSpawner>(FindObjectsSortMode.None)) spawner.enabled = false;
        SessionState.SetBool(Pending,true); EditorApplication.EnterPlaymode();
    }
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    static void IsolateSaves()
    {
        if (!Application.isBatchMode || !SessionState.GetBool(Pending,false)) return;
        string path=Path.Combine(Out,"isolated-progress.json");
        foreach(var suffix in new[]{"",".bak",".tmp"}) if(File.Exists(path+suffix)) File.Delete(path+suffix);
        GameProgressSave.StartFresh=true;
        foreach(var actor in Object.FindObjectsByType<Player>(FindObjectsSortMode.None))
        {
            var save=actor.GetComponent<GameProgressSave>();
            save.StoragePath=path; save.enabled=true;
        }
    }
    static void Changed(PlayModeStateChange state)
    {
        if (state != PlayModeStateChange.EnteredPlayMode || !SessionState.GetBool(Pending,false)) return;
        SessionState.SetBool(Pending,false); results.Clear(); errors.Clear(); steps.Clear(); steps.Push(Checks());
        next = EditorApplication.timeSinceStartup + .3; deadline = EditorApplication.timeSinceStartup + 360;
        Application.logMessageReceived += Log; EditorApplication.update += Tick;
    }
    static void Log(string text,string stack,LogType type)
    {
        if (stack.Contains("UnityEditor.Search") || text.Contains("burst JIT compiler")) return;
        if (type == LogType.Error || type == LogType.Exception || type == LogType.Assert) errors.Add(text + "\n" + stack);
    }
    static void Tick()
    {
        if (EditorApplication.timeSinceStartup < next) return;
        try
        {
            if (EditorApplication.timeSinceStartup > deadline) throw new TimeoutException("Play Mode review timeout");
            while (steps.Count > 0)
            {
                var step = steps.Peek();
                if (!step.MoveNext()) { steps.Pop(); continue; }
                if (step.Current is IEnumerator nested) { steps.Push(nested); continue; }
                next = EditorApplication.timeSinceStartup + (step.Current is float seconds ? seconds : .05); return;
            }
            Check(errors.Count == 0,"No unexpected game/runtime errors: " + string.Join("\n",errors)); Finish(0);
        }
        catch (Exception ex) { results.Add("FAIL " + ex); Finish(1); }
    }
    static void Finish(int code)
    {
        File.WriteAllText(Path.Combine(Out,"key-removal-result.txt"),(code == 0 ? "PASS" : "FAIL") + "\n" + string.Join("\n",results) + "\n" + string.Join("\n",errors));
        Application.logMessageReceived -= Log; EditorApplication.update -= Tick; EditorApplication.Exit(code);
    }
    static void Check(bool value,string message) { if (!value) throw new Exception(message); results.Add("PASS " + message); File.WriteAllLines(Path.Combine(Out,"key-removal-live.txt"),results); }
    static object Get(object target,string field) => target.GetType().GetField(field,Flags).GetValue(target);
    static void Set(object target,string field,object value) => target.GetType().GetField(field,Flags).SetValue(target,value);
    static object Call(object target,string method,params object[] args) => target.GetType().GetMethod(method,Flags).Invoke(target,args);
    static void CloseUpgrade()
    {
        int guard = 0; while (health.IsChoosingUpgrade && guard++ < 24) health.SelectCoopUpgrade(0);
        Time.timeScale = 1;
    }
    static void Move(Vector3 ground)
    {
        if (!player.TryGetTeleportPosition(ground,Quaternion.identity,out var pose)) throw new Exception("Unsafe test teleport: " + ground);
        player.Teleport(pose,Quaternion.identity); Physics.SyncTransforms();
    }
    static IEnumerator KillEncounter(TacticalExpedition expedition,int id)
    {
        var markers = (TacticalMarker[])Get(expedition,"markers");
        Move(markers[id].transform.position);
        yield return .3f; CloseUpgrade();
        if (expedition.ActiveEncounter < 0) expedition.TryBeginEncounter(id,player);
        Check(expedition.ActiveEncounter == id,"Actual Map1 encounter starts: " + id);
        for (int iteration = 0; iteration < 30 && !expedition.Progress.completed[id]; iteration++)
        {
            foreach (var enemy in expedition.EncounterEnemies.ToArray())
            {
                if (enemy == null || enemy.IsDead) continue;
                var boss = enemy.GetComponent<BossController>();
                if (boss != null && boss.IsTransforming) continue;
                enemy.TakeDamage(100000,health,enemy.transform.position - enemy.transform.forward*3);
            }
            Set(expedition,"nextWave",Time.time - 1);
            yield return id == 2 ? 1f : .25f; CloseUpgrade();
        }
        Check(expedition.Progress.completed[id],"Real death paths complete encounter and all waves: " + id);
    }

    {

        player = Object.FindFirstObjectByType<Player>(); health = player.GetComponent<PlayerHealth>(); shooter = player.GetComponent<PlayerShooter>();
        inventory = player.GetComponent<WeaponInventory>(); tactics = player.GetComponent<RogueTactics>();

        Check(expedition != null && expedition.Ready,"Actual Map1 bootstraps tactical encounters on saved meadow/forest geometry");
        Check(Application.companyName=="CodexIsolatedReview" && (string)Get(player.GetComponent<GameProgressSave>(),"sessionPath")==Path.Combine(Out,"isolated-progress.json"),"Both persistent-data directory and active save path are isolated from user progress");

        Set(health,"invincibleUntil",Time.time+300f);
        expedition.enabled=false;
        yield return RemovalDropChecks();
        yield return RemovalChestChecks();
        expedition.enabled=true;
        yield return KillEncounter(expedition,0);
        Check(inventory.Armors.Any(item => item.affix == TacticalAffix.SlideMark),"First encounter awards usable slide-mark boots");
        yield return KillEncounter(expedition,1);
        Check(inventory.Armors.Any(item => item.affix == TacticalAffix.LongEcho),"Second encounter awards echo-range tradeoff pants");
        yield return KillEncounter(expedition,2);
        Check(!DungeonVictoryUI.IsOpen,"Practice boss death does not trigger main campaign victory");
        Check(FloorProgressionManager.Instance.IsComplete(1),"Grass course guarantees all required first-region keys");
        int keys = FloorProgressionManager.Instance.CollectedFor(1);
        var markers = (TacticalMarker[])Get(expedition,"markers");
        Check(!expedition.TryBeginEncounter(0,player) && FloorProgressionManager.Instance.CollectedFor(1) == keys,"Completed event cannot be replayed for duplicate key rewards");

        Check(expedition.Progress.merchantRescued && FloorProgressionManager.Instance.IsComplete(2),"Forest rescue records merchant consequence and guarantees second-region keys");
        Check(Object.FindObjectsByType<TacticalMarker>(FindObjectsSortMode.None).Any(m => m.name.Contains("補給攤")),"Rescued merchant visibly establishes supplies in the next region");
        yield return KillEncounter(expedition,4);
        Check(expedition.Progress.treeRescued && FloorProgressionManager.Instance.IsComplete(3),"Ancient-tree rescue guarantees boss progression");
        var finalBoss = Object.FindObjectsByType<BossController>(FindObjectsSortMode.None).FirstOrDefault(b => b.GetComponent<TacticalTrialBoss>() == null);
        Check(finalBoss != null,"Main boss spawns through the real progression manager");
        Check(Object.FindObjectsByType<TacticalField>(FindObjectsSortMode.None).Any(f=>f.Type==TacticalField.Kind.Shelter),"Tree rescue creates visible protection in the final boss arena");
        Check(finalBoss.GetComponent<TacticalAltarHazard>() != null,"Unbroken altar adds a distinct final-fight danger");
        yield return KillEncounter(expedition,5);
        Set(finalBoss.GetComponent<TacticalAltarHazard>(),"next",Time.time-1); yield return .2f;
        Check(expedition.Progress.altarDestroyed && finalBoss.GetComponent<TacticalAltarHazard>() == null,"Destroying the altar removes its boss hazard");
        var save = player.GetComponent<GameProgressSave>(); yield return .5f;
        Check(save.SaveNow(),"Actual save writes safely to the isolated review path");
        Check(GameProgressSave.TryRead(Path.Combine(Out,"isolated-progress.json"),out var saved,out var error),"Saved expedition decodes: " + error);
        Check(saved.expedition.completed.All(value=>value) && saved.health.shotStyle != TacticalShotStyle.None,"Save preserves event choices and tactical attack selection");
        var cloned = JsonUtility.FromJson<ExpeditionProgress>(JsonUtility.ToJson(saved.expedition));
        expedition.RestoreProgress(cloned);
        Check(expedition.Progress.completed.All(value=>value) && expedition.Progress.merchantRescued && expedition.Progress.treeRescued,"Reload restores visible cross-region consequences");

        foreach (var enemy in EnemyHealth.ActiveEnemies.ToArray()) if (enemy != null) Object.DestroyImmediate(enemy.gameObject);
        foreach (var field in Object.FindObjectsByType<TacticalField>(FindObjectsSortMode.None)) Object.DestroyImmediate(field.gameObject);
        yield return ArenaChecks();
    }
    static void ClearLoot()
    {
        foreach(var item in Object.FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if(item is IMagneticPickup) Object.DestroyImmediate(item.gameObject);
    }
    static IEnumerator RemovalDropChecks()
    {
        ClearLoot(); var original=inventory.CaptureProgress(); var progress=FloorProgressionManager.Instance;
        var originalKeys=progress.CaptureProgress(); var legacyWorld=(int[])originalKeys.Clone(); legacyWorld[1]=9;
        progress.RestoreProgress(legacyWorld);
        var legacy=JsonUtility.FromJson<InventoryProgress>(JsonUtility.ToJson(original)); legacy.keys[1]=1; legacy.keys[2]=2; legacy.keys[3]=3;
        inventory.RestoreProgress(legacy); yield return .1f;
        var normalized=inventory.CaptureProgress();
        Check(normalized.keys[1]==0 && normalized.keys[2]==2 && normalized.keys[3]==3,"Old save removes exactly the pictured first-floor inventory stack; later-floor stacks remain");
        Check(progress.CollectedFor(1)==9,"Removing the inventory item preserves shared keystone progress");
        Check(inventory.Weapons.Count==original.weapons.Length && inventory.Gold==original.gold && inventory.CaptureProgress().healthPacks==original.healthPacks,"Legacy inventory migration preserves weapons, currency and health packs");
        Check(!inventory.GetComponentsInChildren<Text>(true).Any(t=>t.text.Contains("第1層道具")),"Actual inventory UI has no pictured first-floor item slot after restoring old progress");
        inventory.AddFloorKey(1);
        Check(inventory.CaptureProgress().keys[1]==0,"Legacy pickup path cannot add the retired item back to inventory");
        var save=player.GetComponent<GameProgressSave>(); Check(save.SaveNow(),"Migrated inventory writes to isolated save");
        Check(GameProgressSave.TryRead(Path.Combine(Out,"isolated-progress.json"),out var saved,out var error) && saved.inventory.keys[1]==0 && saved.floorKeys[1]==9,"Serialized save removes the item while retaining keystone progress");
        inventory.RestoreProgress(original); progress.RestoreProgress(originalKeys); yield return .1f;
        var defaults=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/NPC/Enemy.prefab").GetComponent<EnemyHealth>();
        Check(Mathf.Approximately(defaults.weakNpcWeaponDropChance,.175f) && Mathf.Approximately(defaults.strongNpcWeaponDropChance,.04f),"Earlier weapon-drop halving remains 17.5% and 4%");
        for(int floor=1;floor<=3;floor++)
        {
            ClearLoot(); int xp=health.CurrentExperience;
            var node=Object.Instantiate(defaults.gameObject,player.transform.position+Vector3.right*50,Quaternion.identity);
            foreach(var behaviour in node.GetComponents<MonoBehaviour>()) if(!(behaviour is EnemyHealth)) behaviour.enabled=false;
            var enemy=node.GetComponent<EnemyHealth>(); enemy.SetDungeonFloor(floor); enemy.SetLevelAndExperience(1,7); enemy.showDamageNumbers=false;
            enemy.weakNpcWeaponDropChance=enemy.strongNpcWeaponDropChance=enemy.armorEquipmentDropChance=enemy.healthDropChance=enemy.progressionDropChance=1;
            enemy.TakeDamage(100000,health);
            Check(enemy.IsDead,"Real production NPC death on floor " + floor);
            Check(Object.FindObjectsByType<FloorKeyPickup>(FindObjectsSortMode.None).Length==(floor==1?0:1),"Floor " + floor + " keystone drop is " + (floor==1?"removed even at 100% configured chance":"retained"));
            Check(Object.FindObjectsByType<WeaponPickup>(FindObjectsSortMode.None).Length==1 && Object.FindObjectsByType<ArmorPickup>(FindObjectsSortMode.None).Length==1 &&
                Object.FindObjectsByType<HealthPickup>(FindObjectsSortMode.None).Length==1 && Object.FindObjectsByType<ExperiencePickup>(FindObjectsSortMode.None).Length==1,"All four other loot categories remain on floor " + floor);
            Check(health.CurrentExperience==xp,"Experience remains a ground pickup on floor " + floor);
            Check(Object.FindFirstObjectByType<ExperiencePickup>().TryCollect(player.GetComponent<Collider>()),"Real XP pickup can be collected on floor " + floor);
            yield return .1f; CloseUpgrade(); ClearLoot();
        }
    }
    static IEnumerator RemovalChestChecks()
    {
        var director=Object.FindFirstObjectByType<TreasureChestDirector>(); director.enabled=false; director.RefreshRooms();
        var rooms=director.GetComponentsInChildren<MedievalDungeonKit.DungeonMarker>().Select(DungeonRoomBoundary.GetOrCreate).Where(r=>r!=null).ToArray();
        for(int floor=1;floor<=3;floor++)
        {
            bool spawned=false;
            foreach(var room in rooms.Where(r=>OverworldDirector.MarkerFloor(r.transform)==floor))
            {
                if(!room.TryGetNavMeshPoint(room.transform.position,1,new NavMeshQueryFilter{agentTypeID=0,areaMask=NavMesh.AllAreas},out var point))continue;
                Move(point); yield return .1f; CloseUpgrade();
                for(int attempt=0;attempt<5 && !spawned;attempt++) spawned=director.TrySpawnChest();
                if(spawned)break;
            }
            Check(spawned,"Actual Map1 floor " + floor + " can spawn a chest");
            var chest=Object.FindFirstObjectByType<TreasureChest>(); chest.TakeDamage(100000,health);
            Check(chest.IsBroken && Object.FindObjectsByType<WeaponPickup>(FindObjectsSortMode.None).Length==1,"Actual floor " + floor + " chest still drops a weapon when broken");
            Object.DestroyImmediate(chest.gameObject); ClearLoot();
        }
    }
    static EnemyHealth Enemy(Vector3 point,float hp=1000)
    {
        var node = GameObject.CreatePrimitive(PrimitiveType.Cube); node.name = "Review Target"; node.transform.position = point;
        node.transform.localScale = new Vector3(1,1.5f,1);
        var enemy = node.AddComponent<EnemyHealth>(); enemy.SetMaxHealth(hp); enemy.SetLevelAndExperience(1,0);
        enemy.healthDropChance=enemy.weakNpcWeaponDropChance=enemy.strongNpcWeaponDropChance=enemy.armorEquipmentDropChance=enemy.progressionDropChance=0;
        enemy.showDamageNumbers = false; Physics.SyncTransforms(); return enemy;
    }
    static void ClearTargets()
    {
        foreach (var enemy in EnemyHealth.ActiveEnemies.ToArray()) if (enemy != null) Object.DestroyImmediate(enemy.gameObject);
        foreach (var shot in Object.FindObjectsByType<Projectile>(FindObjectsSortMode.None)) if (shot != prefab) Object.DestroyImmediate(shot.gameObject);
        foreach (var field in Object.FindObjectsByType<TacticalField>(FindObjectsSortMode.None)) Object.DestroyImmediate(field.gameObject);
        tactics.ResetTransientState();
    }

        boundary=new GameObject("Tactical Arena Boundary").AddComponent<DungeonRoomBoundary>(); boundary.transform.position=arena; boundary.halfSize=Vector2.one*38;
        Move(arena); player.enabled=false; player.GetComponent<Rigidbody>().isKinematic=true;

        foreach (var behaviour in player.GetComponentsInChildren<MonoBehaviour>())
            if (behaviour is FPSCameraLook || behaviour is CameraFollowTPS || behaviour is CameraSwitch) behaviour.enabled=false;
        camera = new GameObject("Tactical Review Camera",typeof(Camera)).GetComponent<Camera>(); camera.tag="MainCamera";

        tactics.AddResonance(100); Record(arena+Vector3.up,Vector3.forward); Check(tactics.TryReleaseEcho(),"Second planned ambush arms");
        player.transform.position=arena+Vector3.right*40+Vector3.up;
        Check(!tactics.TryReleaseEcho() && tactics.EchoArmed,"Out-of-range trigger is rejected without spending the pinned echo");

        tactics.AddResonance(100); Record(arena+Vector3.up,Vector3.forward); tactics.TryReleaseEcho();

    static IEnumerator ExtendedCombatChecks()
    {
        ClearTargets(); player.transform.position=arena+Vector3.up; player.GetComponent<Rigidbody>().position=player.transform.position;
        camera.transform.SetPositionAndRotation(arena+Vector3.up,Quaternion.identity);
        var data=health.CaptureProgress(); data.shotStyle=TacticalShotStyle.Split; health.RestoreProgress(data); CloseUpgrade();
        var bow=inventory.Weapons.First(item=>item.weaponType==WeaponType.Longbow); inventory.EquipToSlot(bow,0); Call(inventory,"SwitchToSlot",0);
        shooter.Fire();
        var shots=Object.FindObjectsByType<Projectile>(FindObjectsSortMode.None).Where(p=>p!=prefab).ToArray();

        Object.DestroyImmediate(image); render.Release(); Object.DestroyImmediate(render);
    }
